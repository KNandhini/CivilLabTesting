function drawCheckbox(doc: PDFDocument, x: number, y: number, checked: boolean) {
  // Set black color and increased stroke width for checkbox border
  const originalStrokeColor = '#A0A0A0';
  const originalLineWidth = 0.5
  doc.strokeColor('black');
  doc.lineWidth(0.8); // Increased stroke width for checkboxes
  
  doc.rect(x, y, 10, 10).stroke();
  if (checked) {
    doc.moveTo(x + 2, y + 5).lineTo(x + 5, y + 8).lineTo(x + 8, y + 2).stroke();
  }
  
  // Reset to original color and line width
  doc.strokeColor(originalStrokeColor);
  doc.lineWidth(originalLineWidth);
}

function getTextHeight(doc: PDFDocument, text: string, options: any) {
  const { width, fontSize, font } = options;
  doc.font(font || 'Calibri').fontSize(fontSize || 10);
  return doc.heightOfString(text, { width });
}
